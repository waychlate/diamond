\begin{minipage}[t]{0.53\textwidth}
    \textbf{\textcolor{teaNavy}{\normalsize Key Findings}}
    \vspace{0.2em}
    
    \small
    \begin{itemize}\setlength{\itemsep}{0.25em}\setlength{\leftskip}{-1.2em}
        \item \textbf{Primed Context Performance}: With 20 frames of historical context, the model achieves near-zero error at Step $+1$ ($100\text{ ms}$ ahead) with $\text{MSE} = \mathbf{0.0255\text{ s}^2}$ ($\text{MAE} = \mathbf{0.13\text{ s}}$).
        \item \textbf{High-Fidelity Control Window}: Across Steps $+1$ to $+5$ ($0.1\text{s} - 0.5\text{s}$ ahead), the model maintains high accuracy ($\text{MAE} < 0.74\text{s}$), suitable for reactive safety interventions.
        \item \textbf{Horizon Accumulation}: In unconditioned deep rollouts ($> 6$ steps), world-model visual dreaming gradually accumulates error, leading to the aggregate MSE rise.
    \end{itemize}
\end{minipage}
